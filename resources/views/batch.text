INSERT INTO ghs_32.profiles (id, user_id,
        name,
        short_name,
        father_name,
        cnic,
        dob,
        blood_group,
        address,
        phone,
        joined_at,
        designation,
        qualification,
        bps,
        personal_no,
        seniority,
        photo
        )
SELECT 
	t.id,
    t.user_id,
    t.name,
    t.short_name,
    t.father_name,
    t.cnic,
    t.dob,
    t.blood_group,
    t.address,
    t.phone,
    t.joined_at,
    t.designation,
    t.qualification,
    t.bps,
    t.personal_no,
    1,
    t.photo
FROM temp.teachers t
JOIN temp.users u 
    ON u.id = t.user_id;

INSERT INTO ghs_32.allocations (id, user_id, subject_id, section_id,lecture_no)
SELECT
    a.id,
    t.user_id,
    a.subject_id,
    a.section_id,
    a.lecture_no
FROM temp.allocations a
JOIN temp.teachers t 
    ON a.teacher_id = t.id;

INSERT INTO ghs_32.test_allocations (id, 
                                     user_id,

                                     subject_id,
                                     section_id, 
                                     test_date, 
                                     max_marks, 
                                     result_date, 
                                     lecture_no, 
                                     created_at, 
                                     updated_at)
SELECT 
    ta.id,
    t.user_id,
    ta.test_id, 
    ta.subject_id,
    ta.section_id, 
    ta.test_date,  
    ta.max_marks, 
    ta.result_date, 
    ta.lecture_no,

    ta.updated_at
